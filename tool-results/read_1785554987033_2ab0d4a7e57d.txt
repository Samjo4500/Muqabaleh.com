   339→
   340→function BeforeAfterSection({ t, isRTL }: { t: ReturnType<typeof useTranslations>; isRTL: boolean }) {
   341→  const containerRef = useRef<HTMLDivElement>(null);
   342→  const [pct, setPct] = useState(50);
   343→
   344→  const handleDrag = useCallback(
   345→    (_: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
   346→      const rect = containerRef.current?.getBoundingClientRect();
   347→      if (!rect) return;
   348→      const raw = ((info.point.x - rect.left) / rect.width) * 100;
   349→      setPct(Math.min(85, Math.max(15, raw)));
   350→    },
   351→    [],
   352→  );
   353→
   354→  return (
   355→    <section id="transform" className="py-20">
   356→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   357→        <SectionHeading
   358→          eyebrow=""
   359→          title={t('whyTitle')}
   360→          titleHighlight={t('whyTitle')}
   361→          sub=""
   362→        />
   363→        <div className="relative mx-auto mt-12 max-w-4xl" ref={containerRef}>
   364→          {/* Desktop: side by side */}
   365→          <div className="hidden gap-4 md:grid md:grid-cols-2 md:gap-0">
   366→            {/* Before card */}
   367→            <div className="glass-card relative overflow-hidden rounded-2xl p-6" style={{ clipPath: `inset(0 ${100 - pct}% 0 0)` }}>
