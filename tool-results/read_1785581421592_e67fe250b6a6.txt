    10→let _geminiModel: InstanceType<import('@google/generative-ai').GoogleGenerativeAI>['getGenerativeModel'] extends (model: string) => infer R ? R : never> | null = null;
    11→let _geminiClient: import('@google/generative-ai').GoogleGenerativeAI | null = null;

    13→async function getGeminiModel() {
    14→  if (_geminiModel) return _geminiModel;
    15→  const apiKey = process.env.GEMINI_API_KEY;
    16→  if (!apiKey) return null;
    17→  try {
    18→    const { GoogleGenerativeAI } = await import('@google/generative-ai');
    19→    _geminiClient = new GoogleGenerativeAI(apiKey);
    20→    _geminiModel = _geminiClient.getGenerativeModel({ model: 'gemini-1.5-flash' });
    21→    return _geminiModel;
    22→  } catch {
    23→    return null;
    24→  }
    25→}

    27→// ─── ZAI Fallback Setup ───
    28→let _zai: Awaited<ReturnType<typeof import('z-ai-web-dev-sdk').default.create>> | null = null;
    29→async function getZAI() {
    30→  if (!_zai) {
    31→    const ZAI = (await import('z-ai-web-dev-sdk')).default;
    32→    _zai = await ZAI.create();
    33→  }
    34→  return _zai;
    35→}

    37→// ─── Unified LLM Call ───
    38→// Tries Gemini first, falls back to z-ai-web-dev-sdk
    39→async function callLLM(messages: { role: string; content: string }[]): Promise<string> {
    40→  // Try Gemini first
    41→  const geminiModel = await getGeminiModel();
    42→  if (geminiModel) {
    43→    try {
    44→      // Gemini format: first message with role=system is systemInstruction,
    45→      // then alternating user/model messages
    46→      let systemInstruction: string | undefined;
    47→      const geminiHistory: { role: 'user' | 'model'; parts: { text: string }[] }[] = [];
    48→      let lastUserMsg = '';
    49→
    50→      for (const msg of messages) {
    51→        if (msg.role === 'system' || msg.role === 'assistant') {
    52→          // Our convention: first 'assistant' message is the system prompt
    53→          if (!systemInstruction && geminiHistory.length === 0) {
    54→            systemInstruction = msg.content;
    55→          } else {
    56→            geminiHistory.push({ role: 'model', parts: [{ text: msg.content }] });
    57→          }
    58→        } else {
    59→          geminiHistory.push({ role: 'user', parts: [{ text: msg.content }] });
    60→          lastUserMsg = msg.content;
    61→        }
    62→      }
    63→
    64→      // If there's a trailing user message, remove it from history (Gemini sends it as prompt)
    65→      if (geminiHistory.length > 0 && geminiHistory[geminiHistory.length - 1].role === 'user') {
    66→        lastUserMsg = geminiHistory.pop()!.parts[0].text;
    67→      }
    68→
    69→      const request: Record<string, unknown> = {
    70→        contents: geminiHistory.length > 0 ? geminiHistory : [{ role: 'user', parts: [{ text: lastUserMsg }] }],
    71→      };
    72→      if (systemInstruction) {
    73→        request.systemInstruction = { parts: [{ text: systemInstruction }] };
    74→      }
    75→
    76→      const result = await geminiModel.generateContent(request);
    77→      const response = result.response;
    78→      const text = response.text();
    79→      if (text) return text;
    80→    } catch (err) {
    81→      console.error('Gemini LLM failed, falling back to ZAI:', err);
    82→    }
    83→  }
    84→
    85→  // Fallback: z-ai-web-dev-sdk
    86→  const zai = await getZAI();
    87→  const completion = await zai.chat.completions.create({
    88→    messages,
    89→    thinking: { type: 'disabled' },
    90→  });
    91→  return completion.choices[0]?.message?.content || '';
    92→}

   205→// ─── Generate Next Question / Continue Interview ───
   206→export async function generateInterviewResponse(
   207→  interviewId: string,
   208→  candidateMessage: string,
   209→  params: InterviewParams,
   210→): Promise<QuestionResult> {
   211→  const dbMessages = await db.message.findMany({
   212→    where: { interviewId },
   213→    orderBy: { sequence: 'asc' },
   214→  });
   215→
   216→  const messageRows: MessageRow[] = dbMessages.map(m => ({
   217→    role: m.role,
   218→    content: m.content,
   219→    sequence: m.sequence,
   220→  }));
   221→
   222→  const questionCount = countInterviewerQuestions(messageRows);
   223→
   224→  if (questionCount >= TOTAL_QUESTIONS) {
   225→    return { question: '', questionNumber: TOTAL_QUESTIONS, totalQuestions: TOTAL_QUESTIONS, done: true };
   226→  }
   227→
   228→  const systemPrompt = buildSystemPrompt(params);
   229→  const questionBank = await fetchQuestionBank(params.industry, params.type, params.language);
   230→  const bankContext = questionBank.length > 0
   231→    ? (params.language === 'AR'
   232→      ? `\n\n\u0628\u0646\u0643 \u0623\u0633\u0626\u0644\u0629 \u0627\u0644\u0645\u0631\u062c\u0639\u064a (\u0627\u0633\u062a\u0644\u0647\u0645 \u0645\u0646\u0647\u0627 \u062f\u0648\u0646 \u062a\u0643\u0631\u0627\u0631\u0647\u0627 \u062d\u0631\u0641\u064a\u0627\u064b):\n${questionBank.map((q, i) => `${i + 1}. ${q}`).join('\n')}`
   233→      : `\n\nReference question bank (draw inspiration without verbatim repetition):\n${questionBank.map((q, i) => `${i + 1}. ${q}`).join('\n')}`)
   234→    : '';
   235→
   236→  const llmMessages: { role: string; content: string }[] = [
   237→    { role: 'assistant', content: systemPrompt + bankContext },
   238→  ];
   239→
   240→  for (const msg of messageRows) {
   241→    llmMessages.push({
   242→      role: msg.role === 'INTERVIEWER' ? 'assistant' : 'user',
   243→      content: msg.content,
   244→    });
   245→  }
   246→  llmMessages.push({ role: 'user', content: candidateMessage });
   247→
   248→  const responseText = await callLLM(llmMessages);
   249→
   250→  const isDone = responseText.includes('[INTERVIEW_DONE]');
   251→  const cleaned = responseText.replace(/\[INTERVIEW_DONE\]/g, '').trim();
   252→
   253→  const nextSeq = (dbMessages[dbMessages.length - 1]?.sequence || 0) + 1;
   254→
   255→  await db.message.create({
   256→    data: { interviewId, role: 'CANDIDATE', content: candidateMessage, sequence: nextSeq },
   257→  });
   258→  await db.message.create({
   259→    data: { interviewId, role: 'INTERVIEWER', content: cleaned, sequence: nextSeq + 1 },
   260→  });
   261→
   262→  return {
   263→    question: cleaned,
   264→    questionNumber: questionCount + 1,
   265→    totalQuestions: TOTAL_QUESTIONS,
   266→    done: isDone,
   267→  };
   268→}

   392→function parseEvaluationJson(raw: string): EvaluationResult | null {
   393→  try {
   394→    const jsonMatch = raw.match(/\{[\s\S]*\}/);
   395→    if (!jsonMatch) return null;
   396→    const parsed = JSON.parse(jsonMatch[0]);
   397→
   398→    const overallScore = clamp(parsed.overallScore, 0, 100);
   399→    const contentScore = clamp(parsed.contentScore, 0, 100);
   400→    const clarityScore = clamp(parsed.clarityScore, 0, 100);
   401→    const confidenceScore = clamp(parsed.confidenceScore, 0, 100);
   402→    const culturalFitScore = clamp(parsed.culturalFitScore, 0, 100);
   403→
   404→    if (parsed.feedback && parsed.strengths && parsed.improvements && parsed.recommendation) {
   405→      return {

   407→        feedback: String(parsed.feedback),

   409→        improvements: Array.isArray(parsed.improvements) ? parsed.improvements.map(String) : [String(parsed.improvements)],
   410→        recommendation: ['RECOMMENDED', 'CONSIDER', 'NOT_RECOMMENDED'].includes(parsed.recommendation)
   411→          ? parsed.recommendation : 'CONSIDER',
   412→      };
   413→    }
   414→  } catch {
   415→    // JSON parse failed
   416→  }
   417→  return null;
   418→}

   420→function heuristicEvaluation(messages: MessageRow[], language: 'AR' | 'EN'): EvaluationResult {
   421→  const candidateMessages = messages.filter(m => m.role === 'CANDIDATE');
   422→  const avgLength = candidateMessages.reduce((sum, m) => sum + m.content.length, 0) / Math.max(candidateMessages.length, 1);
   423→  const lengthScore = Math.min(100, (avgLength / 20) * 100);
   424→  const countScore = Math.min(100, (candidateMessages.length / 7) * 100);
   425→  const overallScore = Math.round((lengthScore * 0.6 + countScore * 0.4));
   426→
   427→  const feedback = language === 'AR'
   428→    ?    429→    : 'This evaluation was automatically generated based on basic answer analysis. Consider conducting another interview for a more accurate assessment.';
   430→
   431→  return {
   432→    overallScore,

   457→// ─── TTS (Text-to-Speech) ───
   458→// Primary: Azure Speech Services (ar-SA-Zariyah / ar-SA-Hamed)
   459→// Fallback: z-ai-web-dev-sdk
   460→const ttsCache = new Map<string, Buffer>();
   461→
   462→export async function textToSpeech(text: string, voice: 'fahd' | 'noora'): Promise<Buffer | null> {
   463→  try {
   464→    const cacheKey = `${voice}:${text}`;
   465→    if (ttsCache.has(cacheKey)) return ttsCache.get(cacheKey)!;
   466→
   467→    const truncated = text.length > 1024 ? text.slice(0, 1020) + '...' : text;
   468→    let buffer: Buffer | null = null;
   469→
   470→    // Try Azure Speech first
   471→    const azureKey = process.env.AZURE_SPEECH_KEY;
   472→    const azureRegion = process.env.AZURE_SPEECH_REGION || 'eastus';
   473→    if (azureKey) {
   474→      try {
   475→        const azureVoice = voice === 'fahd' ? 'ar-SA-Zariyah' : 'ar-SA-Hamed';
   476→        const ssml = `<speak version='1.0' xml:lang='ar-SA'><voice name='${azureVoice}'>${truncated.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')}</voice></speak>`;
   477→
   478→        const res = await fetch(
   479→          `https://${azureRegion}.tts.speech.microsoft.com/cognitiveservices/v1`,
   480→          {
   481→            method: 'POST',
   482→            headers: {
   483→              'Ocp-Apim-Subscription-Key': azureKey,
   484→              'Content-Type': 'application/ssml+xml',
   485→              'X-Microsoft-OutputFormat': 'audio-16khz-128kbitrate-mono-mp3',
   486→              'User-Agent': 'Muqabaleh',
   487→            },
   488→            body: ssml,
   489→          },
   490→        );
   491→
   492→        if (res.ok) {
   493→          const arrayBuf = await res.arrayBuffer();
   494→          buffer = Buffer.from(new Uint8Array(arrayBuf));
   495→        }
   496→      } catch (err) {
   497→        console.error('Azure TTS failed, falling back to ZAI:', err);
   498→      }
   499→    }
   500→
   501→    // Fallback: z-ai-web-dev-sdk
   502→    if (!buffer) {
   503→      const zai = await getZAI();
   504→      const response = await zai.audio.tts.create({
   505→        input: truncated,
   506→        voice: voice === 'fahd' ? 'kazi' : 'chuichui',
   507→        speed: 1.0,
   508→        response_format: 'mp3',
   509→        stream: false,
   510→      });
   511→      const arrayBuf = await response.arrayBuffer();
   512→      buffer = Buffer.from(new Uint8Array(arrayBuf));
   513→    }
   514→
   515→    if (ttsCache.size > 100) {
   516→      const firstKey = ttsCache.keys().next().value;
   517→      ttsCache.delete(firstKey);
   518→    }
   519→    ttsCache.set(cacheKey, buffer);
   520→
   521→    return buffer;
   522→  } catch (err) {
   523→    console.error('TTS failed:', err);
   524→    return null;
   525→  }
   526→}

   536→// ─── Rate Limiter (simple in-memory) ───
   537→const rateLimitMap = new Map<string, { count: number; resetAt: number }>();
   538→
   539→export function checkRateLimit(interviewId: string, maxPerHour: number = 20): boolean {
   540→  const now = Date.now();
   541→  const entry = rateLimitMap.get(interviewId);
   542→  if (!entry || now > entry.resetAt) {
   543→    rateLimitMap.set(interviewId, { count: 1, resetAt: now + 3600_000 });
   544→    return true;
   545→  }
   546→  if (entry.count >= maxPerHour) return false;
   547→  entry.count++;
   548→  return true;
   549→}
